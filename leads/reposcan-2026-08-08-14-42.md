        max_usages,
        description
    };

        const res = await actionRequest('api/user/bootstrap-tokens', 'POST', payload);

        // actionRequest already alerts on failure
    }
};

function writeVarint(value) {
    const bytes = [];
    while (value > 127) {
        bytes.push((value & 127) | 128);
        value >>= 7;
    }
    bytes.push(value);
    return bytes;
}
