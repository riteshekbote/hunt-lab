def classify(filepath, line, content):
    # REAL_SECRET: hardcoded credential with signs of life
    # TEST_OR_EXAMPLE: sample/test/fixture data
    # ENDPOINT_LEAK: internal/dev/staging URLs, API endpoints
    # INTERESTING: anything unusual
    
    # Check for GitHub secrets pattern (proper usage)
    if '${{ secrets.' in content or '${{secrets.' in content:
        return "TEST_OR_EXAMPLE"  # Proper secret reference, not a leak
    
    # Check for placeholder patterns
    placeholders = [
        "REPLACE-WITH-YOUR",
        "your_client_secret",
        "your app's client secret",
        "-----BEGIN PRIVATE KEY-----...",
        'username="user", password="password"',
        'token="eyAkaknabna...."',
        'PASSWORD = "password"',
        'preserve-immutable-source-v1',
    ]
    for p in placeholders:
        if p in content:
            return "TEST_OR_EXAMPLE"
    
    # Check for IMDS/metadata endpoints
    imds_endpoints = [
        "169.254.169.254",
        "metadata.google.internal",
        "100.100.100.200",
        "computeMetadata/v1",
    ]
    for ep in imds_endpoints:
        if ep in content:
            return "ENDPOINT_LEAK"
    
    # Check for variable references (not hardcoded)
    if 'request.client_secret' in content:
        return "INTERESTING"  # Code pattern, not hardcoded secret
    
    # Config/state key names
    if '_STATE_KEY_ACCESS_TOKEN' in content or 'kubeconfig_path' in content or 'kubeconfig_context' in content:
        return "INTERESTING"
    
    # CLI output of API key variable
    if 'GOOGLE_API_KEY=' in content and 'google_api_key' in content:
        return "INTERESTING"
    
    return "INTERESTING"
