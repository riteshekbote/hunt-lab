def classify(filepath, line, content):
    content_lower = content.lower()
    
    # REAL_SECRET patterns - actual credentials with entropy
    if re.search(r'(ya29\.[a-zA-Z0-9_-]+|ghp_[a-zA-Z0-9]{36}|sk-[a-zA-Z0-9]{48}|AIza[a-zA-Z0-9_-]{35})', content):
        return "REAL_SECRET"
    
    # TEST_OR_EXAMPLE patterns
    test_patterns = [
        r'password\s*[=:]\s*["\']password["\']',
        r'token\s*[=:]\s*["\']your_access_token["\']',
        r'PASSWORD\s*[=:]\s*["\']password["\']',
        r'secret_key\s*[=:]\s*["\']fuzz test key["\']',
        r'shutdownPassword\s*[=:]\s*["\']ServerShutdownPassword["\']',
        r'private_key\s*[=:]\s*["\']-----BEGIN PRIVATE KEY-----\.\.\.["\']',
        r'username\s*[=:]\s*["\']user["\'].*password\s*[=:]\s*["\']password["\']',
        r'token\s*[=:]\s*["\']eyAkaknabna\.\.\.["\']',
        r'client_secret\s*[=:]\s*["\']your_client_secret["\']',
        r'INJECTED_TOKEN\s*[=:]\s*["\']injected-token-probe["\']',
    ]
    for pat in test_patterns:
        if re.search(pat, content, re.IGNORECASE):
            return "TEST_OR_EXAMPLE"
    
    # ENDPOINT_LEAK - internal/metadata endpoints
    if any(x in content for x in ['169.254.169.254', 'metadata.google.internal', 'metadata.azure.com', 'metadata.aws']):
        return "ENDPOINT_LEAK"
    
    # INTERESTING - config references, secret managers, etc.
    if any(x in content_lower for x in ['set-secrets', 'secret manager', 'vault', 'kubeconfig', 'private-key.pem', 'ADMIN_PASSWORD', 'GRR_MYSQL_PASSWORD']):
        return "INTERESTING"
    
    # Minified JS / build artifacts
    if filepath.endswith('.js') and ('minified' in content_lower or 'chunk-' in filepath or 'main-' in filepath):
        return "TEST_OR_EXAMPLE"  # build artifacts
    
    # Comments, URLs, homepage
    if content_lower.startswith('comment') or content_lower.startswith('homepage') or content_lower.startswith('url'):
        return "TEST_OR_EXAMPLE"
    
    return "INTERESTING"
