def classify(file, line, content):
    content_lower = content.lower()
    
    # REAL_SECRET patterns
    if re.search(r'(api[_-]?key|secret|token|password|private[_-]?key)\s*[:=]\s*["\'][a-zA-Z0-9_\-]{20,}["\']', content):
        return "REAL_SECRET"
    if re.search(r'(ya29\.|eyJ|gh[opsu]_[a-zA-Z0-9]{36,})', content):
        return "REAL_SECRET"
    
    # TEST_OR_EXAMPLE patterns
    if any(x in file for x in ['test', 'Test', 'TEST', 'example', 'Example', 'fixture', 'Fixture']):
        if any(x in content_lower for x in ['test', 'example', 'fake', 'dummy', 'placeholder', 'your_', '...']):
            return "TEST_OR_EXAMPLE"
    if any(x in content_lower for x in ['testpassword', 'test-token', 'test-token-123', 'your_client_secret', 'your_api_key', 'password="password"', 'username="user"', 'token="eyAkaknabna', 'private_key="-----begin private key-----']):
        return "TEST_OR_EXAMPLE"
    if 'PASSWORD = "password"' in content:
        return "TEST_OR_EXAMPLE"
    
    # ENDPOINT_LEAK patterns
    if any(x in content for x in ['corp.google.com', 'devsite.corp.google.com', 'source.corp.google.com', 'pantheon.corp.google.com', 'dev.office.com']):
        return "ENDPOINT_LEAK"
    if re.search(r'https?://(dev|internal|staging|prod)\.', content):
        return "ENDPOINT_LEAK"
    
    # INTERESTING patterns
    if '169.254.169.254' in content:
        return "INTERESTING"
    if 'kubeconfig' in content_lower:
        return "INTERESTING"
    if 'GOOGLE_API_KEY' in content:
        return "INTERESTING"
    if '_STATE_KEY_ACCESS_TOKEN' in content or '_vmaas_access_token' in content:
        return "INTERESTING"
    if 'legacy openssl rsa key format' in content_lower or 'BEGIN RSA PRIVATE KEY' in content:
        return "INTERESTING"
    if '@see http://dev.w3.org' in content or '@see {http://dev.w3.org' in content:
        return "TEST_OR_EXAMPLE"  # Spec references
    
    return "TEST_OR_EXAMPLE"

results = {"REAL_SECRET": [], "TEST_OR_EXAMPLE": [], "ENDPOINT_LEAK": [], "INTERESTING": []}

        if m:
            secret_hash = hashlib.sha256(m.group(1).encode()).hexdigest()[:16]
        else:
            secret_hash = hashlib.sha256(c.encode()).hexdigest()[:16]
    results[cat].append((f, l, c[:100], secret_hash))

for cat in ["REAL_SECRET", "TEST_OR_EXAMPLE", "ENDPOINT_LEAK", "INTERESTING"]:

# Verdict
